    ```yaml
        name: lighthouse
        runs-on: ubuntu-latest
        steps:

            with: { cache: true }
          - uses: actions/setup-node@v7

          - run: pnpm build
          - run: pnpm exec lhci autorun
          - if: always()
            uses: actions/upload-artifact@v7
            with:
              name: lighthouse
              path: .lighthouseci
              retention-days: 7
    ```
