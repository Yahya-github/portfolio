function bindWheel(controller: PagerController): Cleanup {
  let wheel = INITIAL_WHEEL;
  const onWheel = (e: WheelEvent) => {
    const mode = e.deltaMode as 0 | 1 | 2;
    const legacy = (e as WheelEvent & { wheelDeltaY?: number }).wheelDeltaY;
    const sample = { mode, dx: e.deltaX, dy: e.deltaY, t: e.timeStamp, ctrl: e.ctrlKey, vh: window.innerHeight, legacy };
    const edge = edgeInfo(currentScroller(controller));
    const step = stepWheel(wheel, sample, edge);
    wheel = step.state;
    if (step.preventDefault) e.preventDefault();
    if (step.intent) controller.intent({ type: step.intent, source: "wheel" });

}

/** Pre-built in-page wheel stream: every event is constructed before any handler runs, so timestamps never include handler cost. */
async function dispatchWheelStream(page: Page, dy: number, count: number): Promise<void> {
  await page.evaluate(([d, n]) => {

    for (const e of events) target.dispatchEvent(e);
  }, [dy, count] as const);
}
