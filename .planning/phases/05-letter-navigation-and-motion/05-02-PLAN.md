```ts
export const SLIDE_MS = 600;
const FALLBACK_EASE = { page: "cubic-bezier(0.77, 0, 0.18, 1)", retarget: "cubic-bezier(0.3, 0.6, 0.18, 1)" } as const;
function easing(retarget: boolean): string {
  const value = getComputedStyle(document.documentElement).getPropertyValue(retarget ? "--ease-retarget" : "--ease-page").trim();
  return value || (retarget ? FALLBACK_EASE.retarget : FALLBACK_EASE.page);
}
function isCssDriven(animation: Animation): boolean {
  return (typeof CSSAnimation !== "undefined" && animation instanceof CSSAnimation) ||
         (typeof CSSTransition !== "undefined" && animation instanceof CSSTransition);
}
function cancelOn(el: HTMLElement | null): void { el?.getAnimations?.().filter((a) => !isCssDriven(a)).forEach((a) => a.cancel()); }
function canAnimate(el: HTMLElement | null): el is HTMLElement { return el !== null && typeof el.animate === "function"; }
```

    ```ts
    import type { LettersMode } from "@/lib/letters/geometry";
    export interface GlideStart { readonly to: number; readonly retarget: boolean; readonly heldScrollY: number }
    export interface GlideOptions {
      readonly nav: HTMLElement;                 // nav[data-pager="letters"]
      readonly isReduced: () => boolean;         // controller.isReducedMotion (live, WR-08)
      readonly intercept?: (start: GlideStart) => boolean; // true = another module owns this arrival (D-03 opening, plan 09)
    }
    export interface Glide {
      start(start: GlideStart): void;
      snapToRest(): void;      // resize / reduced-motion switch: cancel and write the heading mode now
      heading(): LettersMode;  // mode the letters are at or flying to
      isFlying(): boolean;
      destroy(): void;
    }
    export function createGlide(options: GlideOptions): Glide;
    ```
    Internal state: `heading: LettersMode` initialised from `nav.dataset.mode === "rail" ? "rail" : "hero"`, `flying = false`, `flightId = 0`. `letters()` = `Array.from(nav.querySelectorAll<HTMLElement>(":scope > a"))`. `rects()` maps to `{left, top, height}` from getBoundingClientRect.

    `settle(mode)`: `nav.dataset.mode = mode; delete nav.dataset.flight; letters().forEach(cancelOn); flying = false; heading = mode; document.dispatchEvent(new Event(GLIDE_SETTLED_EVENT));` (mode write, flight removal and cancel in the same task so nothing snaps).

    `start(s)`:
    1. If `options.intercept?.(s)` is true: `flightId += 1; letters().forEach(cancelOn); flying = false; heading = "hero"; return;` (the opening writes the DOM itself).
    2. `const plan = planGlide(heading, s.to)`; if `plan.kind === "none"` return (rail-to-rail and same-destination retargets keep running, D-13).
    3. If `options.isReduced()` or `!canAnimate(letters()[0] ?? null)`: `flightId += 1; settle(plan.dest); return;` (Pitfall 2: WAAPI ignores --dur tokens).
    4. Read rendered rects FIRST: `let first = rects()`. If `nav.dataset.mode === "hero" && s.heldScrollY > 0`, `first = first.map((b) => shiftBox(b, s.heldScrollY))` (Pitfall 6: the absolute hero row moved with the scroll reset that just happened).
    5. `letters().forEach(cancelOn)`; then in one synchronous block: `nav.dataset.mode = "rail"; const rail = rects(); nav.dataset.mode = "hero"; const hero = rects();` and `nav.dataset.flight = plan.dest === "rail" ? "out" : "in";`.
    6. For each letter i (0..4): `from = deltaOnto(hero[i], first[i])`, `to = plan.dest === "rail" ? deltaOnto(hero[i], rail[i]) : IDENTITY`, `{xDelay, yDelay} = letterTiming(i)`. Two animations, both `fill: "both"` (Pitfall 1):
       - `el.animate([{ translate: `${from.dx}px 0px` }, { translate: `${to.dx}px 0px` }], { duration: LETTERS_MS, delay: xDelay, easing: s.retarget ? readEase("--ease-retarget", FALLBACK_EASE.retarget) : readEase("--ease-page", FALLBACK_EASE.page), fill: "both" })`
       - `el.animate([{ transform: `translateY(${from.dy}px) scale(${from.s})` }, { transform: `translateY(${to.dy}px) scale(${to.s})` }], { duration: LETTERS_MS, delay: yDelay, easing: readEase("--ease-settle", FALLBACK_EASE.settle), fill: "both" })`
    7. `const id = ++flightId; flying = true; heading = plan.dest;` then `Promise.all(animations.map((a) => a.finished.then(() => true, () => false))).then(() => { if (id === flightId) settle(plan.dest); });` (stale flights are ignored; a flight cancelled from outside still settles, never strands).

    `snapToRest()`: `if (!flying) return; flightId += 1; settle(heading);`. `destroy()`: `flightId += 1; letters().forEach(cancelOn);` (no attribute writes). Only `translate`, `transform`, `opacity` are ever animated. No timers, no rAF, no logging.
