`components/shell/Track.tsx`:
```tsx
import type { ComponentType } from "react";
import type { Slug } from "@/content/types";
import type { PageProps } from "@/components/pages/types";
import { sections } from "@/content/site";
import { IndexPage } from "@/components/pages/IndexPage";
import { AchievementsPage } from "@/components/pages/AchievementsPage";
import { ProjectsPage } from "@/components/pages/ProjectsPage";
import { ExperiencePage } from "@/components/pages/ExperiencePage";
import { AboutPage } from "@/components/pages/AboutPage";
import { SayHiPage } from "@/components/pages/SayHiPage";
import styles from "./shell.module.css";
const PAGE_COMPONENTS: Readonly<Record<Slug, ComponentType<PageProps>>> = { index: IndexPage, achievements: AchievementsPage, projects: ProjectsPage, experience: ExperiencePage, about: AboutPage, "say-hi": SayHiPage };
export function Track({ current }: { readonly current: Slug }) {
  return (
    <main id="content" tabIndex={-1} data-current={current} className={styles.track}>
      {sections.map((s) => {
        const isCurrent = s.slug === current;
        const Body = PAGE_COMPONENTS[s.slug];
        return (
          <section key={s.slug} id={`page-${s.slug}`} data-page={s.slug} inert={!isCurrent} aria-labelledby={`title-${s.slug}`} className={styles.page}>
            <Body headingLevel={isCurrent ? 1 : 2} isCurrent={isCurrent} />
          </section>
        );
      })}
    </main>
  );
}
```
React 19 serialises `inert={true}` as `inert=""` and omits the attribute when false (verified in RESEARCH).
