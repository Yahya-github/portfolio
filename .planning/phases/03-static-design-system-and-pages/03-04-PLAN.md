2. Create `components/media/Photo.tsx`. It is a server component: no 'use client', no hooks.
```tsx
import type { StaticImageData } from "next/image";
export interface PhotoProps { readonly sources: readonly StaticImageData[]; readonly sizes: string; readonly alt: string; readonly eager: boolean; readonly className?: string; }
export function Photo({ sources, sizes, alt, eager, className }: PhotoProps) {
  const widest = sources.reduce((a, b) => (b.width > a.width ? b : a));
  const srcSet = sources.map((s) => `${s.src} ${s.width}w`).join(", ");
  return (
    // eslint-disable-next-line @next/next/no-img-element -- static export has no optimizer; sources are pre-sized WebP (Pitfall 9)
    <img className={className} src={widest.src} srcSet={srcSet} sizes={sizes} width={widest.width} height={widest.height} alt={alt} loading={eager ? "eager" : "lazy"} decoding="async" />
  );
}
```
This file holds the ONLY `no-img-element` disable in the project (user decision 2026-11-08).

export type HeadingLevel = 1 | 2 | 3 | 4;
const TAGS = { 1: "h1", 2: "h2", 3: "h3", 4: "h4" } as const;
export function Heading({ level, id, className, children }: { readonly level: HeadingLevel; readonly id?: string; readonly className?: string; readonly children: ReactNode }) { const Tag = TAGS[level]; return <Tag id={id} className={className}>{children}</Tag>; }

}

export const DEFAULT_WORD_EM = 4.6;
